## الخطوة 1: `src/components/MosaicsSection.tsx` (استبدال كامل)

صورة عريضة واحدة بحركة موازية (الحركة رقم 2 من 2)، وتحتها أسماء الجداريات كتبويبات نصية.

```tsx
import React, { useMemo, useState } from 'react';
import { motion } from 'motion/react';
import { Language, LightboxImageItem } from '../types';
import { MOSAICS, SECTIONS } from '../data/content';
import { imgAlt, imgSrc } from '../data/images';
import { useLightbox } from '../context/LightboxContext';
import { ParallaxImage } from './Motion';
import { SectionHead } from './SectionHead';

interface MosaicsSectionProps {
  lang: Language;
}

export const MosaicsSection: React.FC<MosaicsSectionProps> = React.memo(({ lang }) => {
  const isAr = lang === 'ar';
  const { openLightbox } = useLightbox();
  const [index, setIndex] = useState(0);
  const current = MOSAICS[index];

  const slides: LightboxImageItem[] = useMemo(
    () =>
      MOSAICS.map((m) => ({
        src: imgSrc(m.image),
        title: m.title[lang],
        caption: m.text[lang],
      })),
    [lang]
  );

  return (
    <section
      id="mosaics"
      aria-labelledby="mosaics-heading"
      className="bg-ivory px-5 py-16 sm:px-8 sm:py-24"
    >
      <div className="mx-auto max-w-6xl">
        <SectionHead
          id="mosaics-heading"
          no="03"
          title={SECTIONS.mosaics.title[lang]}
          lead={SECTIONS.mosaics.lead[lang]}
          isAr={isAr}
        />

        <button
          type="button"
          onClick={() => openLightbox(slides, index)}
          aria-label={isAr ? 'تكبير الصورة' : 'Enlarge image'}
          className="block w-full cursor-zoom-in"
        >
          <ParallaxImage
            src={imgSrc(current.image)}
            alt={imgAlt(current.image, lang)}
            lang={lang}
            className="aspect-[4/3] w-full bg-[#efece6] md:aspect-[21/9]"
          />
        </button>

        <div role="tablist" className="mt-8 flex gap-8 overflow-x-auto border-b border-line">
          {MOSAICS.map((m, i) => (
            <button
              key={m.id}
              type="button"
              role="tab"
              id={`mosaic-tab-${m.id}`}
              aria-selected={i === index}
              aria-controls="mosaic-panel"
              onClick={() => setIndex(i)}
              className={`-mb-px cursor-pointer whitespace-nowrap border-b-2 pb-3 text-lg transition-colors ${
                i === index ? 'border-gold text-ink' : 'border-transparent text-ink-soft hover:text-ink'
              }`}
            >
              {m.tab[lang]}
            </button>
          ))}
        </div>

        <motion.div
          key={current.id}
          id="mosaic-panel"
          role="tabpanel"
          aria-labelledby={`mosaic-tab-${current.id}`}
          initial={{ opacity: 0, y: 8 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.5, ease: [0.22, 1, 0.36, 1] }}
          className="mt-8 grid gap-4 md:grid-cols-[1fr_1.6fr] md:gap-12"
        >
          <h3 className="row-title">{current.title[lang]}</h3>
          <p className="lead">{current.text[lang]}</p>
        </motion.div>
      </div>
    </section>
  );
});
```

## الخطوة 2: `src/components/PhotoGallerySection.tsx` (استبدال كامل)

شريط أفقي بأحجام متنوعة (طويل وعريض ومربع)، مع ظهور بفتح تدريجي لكل صورة، وأسهم للتمرير. بدون فلاتر وبدون شبكة.

```tsx
import React, { useMemo, useRef } from 'react';
import { motion, useReducedMotion } from 'motion/react';
import { ChevronLeft, ChevronRight } from 'lucide-react';
import { Language, LightboxImageItem } from '../types';
import { SECTIONS } from '../data/content';
import { GALLERY_KEYS, IMAGES, imgSrc } from '../data/images';
import { useLightbox } from '../context/LightboxContext';
import { Picture } from './Picture';
import { SectionHead } from './SectionHead';

interface PhotoGallerySectionProps {
  lang: Language;
}

// إيقاع الأحجام: طويل، عريض، مربع، طويل أقصر
const SIZES = [
  'h-72 w-52 sm:h-96 sm:w-64',
  'h-48 w-72 sm:h-64 sm:w-96',
  'h-60 w-60 sm:h-72 sm:w-72',
  'h-64 w-52 sm:h-80 sm:w-60',
];

export const PhotoGallerySection: React.FC<PhotoGallerySectionProps> = React.memo(({ lang }) => {
  const isAr = lang === 'ar';
  const { openLightbox } = useLightbox();
  const reduce = useReducedMotion();
  const scroller = useRef<HTMLDivElement>(null);

  const slides: LightboxImageItem[] = useMemo(
    () =>
      GALLERY_KEYS.map((key) => ({
        src: imgSrc(key),
        title: IMAGES[key][lang],
      })),
    [lang]
  );

  // next = الاتجاه الذي يكشف الصور التالية (يسار في العربية، يمين في الإنجليزية)
  const scrollByStep = (next: boolean) => {
    const el = scroller.current;
    if (!el) return;
    const amount = el.clientWidth * 0.8;
    const toRight = isAr ? !next : next;
    el.scrollBy({ left: toRight ? amount : -amount, behavior: 'smooth' });
  };

  return (
    <section
      id="gallery"
      aria-labelledby="gallery-heading"
      className="overflow-hidden bg-white py-16 sm:py-24"
    >
      <div className="mx-auto max-w-6xl px-5 sm:px-8">
        <SectionHead
          id="gallery-heading"
          no="04"
          title={SECTIONS.gallery.title[lang]}
          lead={SECTIONS.gallery.lead[lang]}
          isAr={isAr}
        />
      </div>

      <div
        ref={scroller}
        tabIndex={0}
        aria-label={isAr ? 'صور المعرض، مرّر أفقياً' : 'Gallery photos, scroll horizontally'}
        className="flex snap-x snap-mandatory items-end gap-4 overflow-x-auto px-5 pb-4 sm:gap-6 sm:px-8"
      >
        {GALLERY_KEYS.map((key, i) => (
          <figure key={key} className="shrink-0 snap-start">
            <button
              type="button"
              onClick={() => openLightbox(slides, i)}
              aria-label={IMAGES[key][lang]}
              className={`group block cursor-zoom-in overflow-hidden bg-[#efece6] ${SIZES[i % SIZES.length]}`}
            >
              <motion.div
                className="h-full w-full"
                initial={reduce ? false : { clipPath: 'inset(0 0 100% 0)' }}
                whileInView={{ clipPath: 'inset(0 0 0% 0)' }}
                viewport={{ root: scroller, once: true, amount: 0.25 }}
                transition={{ duration: 0.9, delay: (i % 3) * 0.08, ease: [0.22, 1, 0.36, 1] }}
              >
                <Picture
                  src={imgSrc(key)}
                  alt={IMAGES[key][lang]}
                  lang={lang}
                  className="h-full w-full transition-transform duration-700 group-hover:scale-[1.04]"
                />
              </motion.div>
            </button>
            <figcaption className="mt-3 text-sm text-ink-soft">{IMAGES[key][lang]}</figcaption>
          </figure>
        ))}
      </div>

      <div className="mx-auto mt-6 flex max-w-6xl gap-2 px-5 sm:px-8">
        <button
          type="button"
          onClick={() => scrollByStep(false)}
          aria-label={isAr ? 'السابق' : 'Previous'}
          className="grid h-11 w-11 cursor-pointer place-items-center border border-line text-ink transition-colors hover:border-gold"
        >
          {isAr ? <ChevronRight className="h-5 w-5" strokeWidth={1.25} /> : <ChevronLeft className="h-5 w-5" strokeWidth={1.25} />}
        </button>
        <button
          type="button"
          onClick={() => scrollByStep(true)}
          aria-label={isAr ? 'التالي' : 'Next'}
          className="grid h-11 w-11 cursor-pointer place-items-center border border-line text-ink transition-colors hover:border-gold"
        >
          {isAr ? <ChevronLeft className="h-5 w-5" strokeWidth={1.25} /> : <ChevronRight className="h-5 w-5" strokeWidth={1.25} />}
        </button>
      </div>
    </section>
  );
});
```

## الخطوة 3: `src/components/VisitSection.tsx` (استبدال كامل)

المداخل كصفوف قابلة للفتح (بدون صور)، وآداب الزيارة بأربعة أعمدة نصية، وأوقات الزيارة بفقرة واحدة. هذا القسم بلا أي صورة.

```tsx
import React, { useState } from 'react';
import { Language } from '../types';
import { CONDUCT, GATES, HOURS, SECTIONS, VISIT_LABELS } from '../data/content';
import { AccordionRow } from './AccordionRow';
import { Reveal } from './Motion';
import { SectionHead } from './SectionHead';

interface VisitSectionProps {
  lang: Language;
}

export const VisitSection: React.FC<VisitSectionProps> = React.memo(({ lang }) => {
  const isAr = lang === 'ar';
  const [openId, setOpenId] = useState<string | null>(null);

  return (
    <section
      id="visit"
      aria-labelledby="visit-heading"
      className="bg-ivory px-5 py-16 sm:px-8 sm:py-24"
    >
      <div className="mx-auto max-w-6xl">
        <SectionHead
          id="visit-heading"
          no="05"
          title={SECTIONS.visit.title[lang]}
          lead={SECTIONS.visit.lead[lang]}
          isAr={isAr}
        />

        <h3 className="sr-only">{VISIT_LABELS.gates[lang]}</h3>
        <div className="border-t border-line">
          {GATES.map((g) => (
            <AccordionRow
              key={g.id}
              id={`gate-${g.id}`}
              eyebrow={g.direction[lang]}
              title={g.name[lang]}
              summary={g.summary[lang]}
              open={openId === g.id}
              onToggle={() => setOpenId(openId === g.id ? null : g.id)}
            >
              <p className="lead max-w-2xl">{g.more[lang]}</p>
            </AccordionRow>
          ))}
        </div>

        <div className="mt-16 sm:mt-20">
          <Reveal>
            <h3 className="row-title mb-8">{VISIT_LABELS.conduct[lang]}</h3>
          </Reveal>
          <Reveal>
            <ul className="grid gap-x-10 gap-y-8 border-t border-line pt-8 sm:grid-cols-2 lg:grid-cols-4">
              {CONDUCT.map((c) => (
                <li key={c.id}>
                  <h4 className="text-lg font-medium text-ink">{c.title[lang]}</h4>
                  <p className="mt-2 text-base font-light leading-8 text-ink-soft">{c.text[lang]}</p>
                </li>
              ))}
            </ul>
          </Reveal>
        </div>

        <Reveal className="mt-14 max-w-3xl border-s-2 border-gold ps-6 sm:mt-16">
          <h3 className="eyebrow">{VISIT_LABELS.hours[lang]}</h3>
          <p className="lead mt-2">{HOURS[lang]}</p>
        </Reveal>
      </div>
    </section>
  );
});
```

## الخطوة 4: الفحص النهائي

1. شغّل `npm run lint`. إذا ظهرت أخطاء، أصلحها **بأقل تعديل ممكن** (استيراد ناقص، خطأ إملائي في اسم). لا تغيّر التصميم ولا النصوص.
2. شغّل `npm run build`. يجب أن ينجح دون أخطاء.
3. ابحث في كل ملفات `src/` عن أي استيراد من الملفات المحذوفة (`SafeImage`، `MosqueProgrammaticArtwork`، `SectionTransitionWrapper`، `AnimatedNumber`، `FormattedText`، `ParallaxStatsSection`) وعن أي استخدام لـ `MOSQUE_IMAGES` أو `MOSAIC_ARTWORKS` أو `TIMELINE_EVENTS` أو `HERO_CONTENT` أو `PHOTO_GALLERY_ITEMS` أو `ETIQUETTE_GUIDELINES`. إن وجدت شيئاً اذكره للمستخدم ولا تحذفه بنفسك.
4. اعرض للمستخدم هذه القائمة، وأجب عن كل بند بـ "نعم" أو "لا" فقط بعد أن تتحقق فعلاً:
   - القائمة العلوية فيها ستة عناصر وتنقلك لكل قسم، والمعرض بينها
   - القائمة العلوية تبقى ثابتة أعلى الشاشة أثناء التمرير
   - التبديل بين العربية والإنجليزية يغيّر كل النصوص والاتجاه
   - لا توجد زوايا مستديرة كبيرة ولا ظلال ثقيلة ولا إطارات ذهبية حول الصور
   - لا توجد أرقام متحركة (عدّاد) في أي مكان
   - خلفية التذييل وحده خضراء داكنة
   - الحركة الموازية للصورة موجودة في موضعين فقط: صورة الرئيسية وصورة الفسيفساء
   - نافذة عرض الصور تفتح من كل الأقسام، وأسهمها تعمل بشكل صحيح بالعربية
   - لا يوجد أي رقم ارتفاع أو طول أو مساحة في المحتوى
5. اكتب للمستخدم تقريراً من خمسة أسطر كحد أقصى.
