## الخطوة 1: `src/App.tsx` (استبدال كامل)

```tsx
import React, { useCallback, useEffect, useRef, useState } from 'react';
import { Language, SectionId } from './types';
import { Header } from './components/Header';
import { HeroSection } from './components/HeroSection';
import { TimelineSection } from './components/TimelineSection';
import { ArchitectureSection } from './components/ArchitectureSection';
import { MosaicsSection } from './components/MosaicsSection';
import { PhotoGallerySection } from './components/PhotoGallerySection';
import { VisitSection } from './components/VisitSection';
import { Footer } from './components/Footer';
import { LightboxProvider } from './context/LightboxContext';

const SECTION_IDS: SectionId[] = ['home', 'history', 'architecture', 'mosaics', 'gallery', 'visit'];

export default function App() {
  const [lang, setLang] = useState<Language>('ar');
  const [activeSection, setActiveSection] = useState<SectionId>('home');
  const locked = useRef(false);
  const lockTimer = useRef<number | null>(null);

  // اتجاه الصفحة واللغة وعنوان التبويب
  useEffect(() => {
    document.documentElement.dir = lang === 'ar' ? 'rtl' : 'ltr';
    document.documentElement.lang = lang;
    document.title =
      lang === 'ar'
        ? 'الجامع الأموي الكبير بدمشق | وزارة الأوقاف'
        : 'The Great Umayyad Mosque of Damascus | Ministry of Endowments';
  }, [lang]);

  // تمييز القسم الظاهر في القائمة أثناء التمرير
  useEffect(() => {
    const observers: IntersectionObserver[] = [];

    SECTION_IDS.forEach((id) => {
      const el = document.getElementById(id);
      if (!el) return;
      const observer = new IntersectionObserver(
        (entries) => {
          if (locked.current) return;
          entries.forEach((entry) => {
            if (entry.isIntersecting) setActiveSection(id);
          });
        },
        { rootMargin: '-20% 0px -60% 0px', threshold: 0 }
      );
      observer.observe(el);
      observers.push(observer);
    });

    return () => {
      observers.forEach((o) => o.disconnect());
      if (lockTimer.current) window.clearTimeout(lockTimer.current);
    };
  }, []);

  const handleNavigate = useCallback((id: SectionId) => {
    setActiveSection(id);
    locked.current = true;
    if (lockTimer.current) window.clearTimeout(lockTimer.current);
    lockTimer.current = window.setTimeout(() => {
      locked.current = false;
    }, 600);

    if (id === 'home') {
      window.scrollTo({ top: 0, left: 0 });
      if (window.location.hash) window.history.replaceState(null, '', window.location.pathname);
      return;
    }

    document.getElementById(id)?.scrollIntoView({ block: 'start' });
    window.history.replaceState(null, '', `#${id}`);
  }, []);

  return (
    <LightboxProvider lang={lang}>
      <div className="flex min-h-screen flex-col bg-white text-ink">
        <a
          href="#main-content"
          className="sr-only focus:not-sr-only focus:fixed focus:start-4 focus:top-4 focus:z-[100] focus:bg-primary focus:px-4 focus:py-2 focus:text-white"
        >
          {lang === 'ar' ? 'تخطَّ إلى المحتوى' : 'Skip to content'}
        </a>

        <Header
          lang={lang}
          onLanguageChange={setLang}
          activeSection={activeSection}
          onNavigate={handleNavigate}
        />

        <main id="main-content" className="flex-1">
          <HeroSection lang={lang} />
          <TimelineSection lang={lang} />
          <ArchitectureSection lang={lang} />
          <MosaicsSection lang={lang} />
          <PhotoGallerySection lang={lang} />
          <VisitSection lang={lang} />
        </main>

        <Footer lang={lang} />
      </div>
    </LightboxProvider>
  );
}
```

## الخطوة 2: `src/components/PrayerWidget.tsx` (استبدال كامل)

```tsx
import React, { useEffect, useMemo, useState } from 'react';
import { Language } from '../types';
import { calculateDamascusPrayerTimes, formatTimeWithPeriod } from '../utils/prayerTimes';

interface PrayerWidgetProps {
  lang: Language;
}

export const PrayerWidget: React.FC<PrayerWidgetProps> = React.memo(({ lang }) => {
  const [stamp, setStamp] = useState<number>(() => Math.floor(Date.now() / 60000));

  useEffect(() => {
    const timer = setInterval(() => setStamp(Math.floor(Date.now() / 60000)), 15000);
    return () => clearInterval(timer);
  }, []);

  const data = useMemo(() => calculateDamascusPrayerTimes(new Date(), lang), [stamp, lang]);
  const isAr = lang === 'ar';

  const items = [
    { id: 'fajr', name: isAr ? 'الفجر' : 'Fajr', time: data.fajr },
    { id: 'sunrise', name: isAr ? 'الشروق' : 'Sunrise', time: data.sunrise },
    { id: 'dhuhr', name: isAr ? 'الظهر' : 'Dhuhr', time: data.dhuhr },
    { id: 'asr', name: isAr ? 'العصر' : 'Asr', time: data.asr },
    { id: 'maghrib', name: isAr ? 'المغرب' : 'Maghrib', time: data.maghrib },
    { id: 'isha', name: isAr ? 'العشاء' : 'Isha', time: data.isha },
  ];

  return (
    <div id="prayer-times-widget">
      <div className="flex flex-wrap items-center justify-between gap-2 py-4">
        <h3 className="text-base font-medium text-ink">
          {isAr ? 'مواقيت الصلاة في دمشق' : 'Prayer times in Damascus'}
        </h3>
        <span className="text-sm text-ink-soft">{data.hijriDate}</span>
      </div>

      <dl className="grid grid-cols-3 md:grid-cols-6">
        {items.map((p) => {
          const next = data.nextPrayer?.id === p.id;
          return (
            <div
              key={p.id}
              className={`border-t-2 px-1 py-5 text-center ${next ? 'border-gold' : 'border-line'}`}
            >
              <dt className={`text-sm ${next ? 'font-medium text-gold-deep' : 'text-ink-soft'}`}>{p.name}</dt>
              <dd className="mt-1 text-xl font-light tabular-nums text-ink sm:text-2xl">
                {formatTimeWithPeriod(p.time, lang)}
              </dd>
            </div>
          );
        })}
      </dl>
    </div>
  );
});
```

## الخطوة 4: `src/components/TimelineSection.tsx` (استبدال كامل)

أربع حقب كصفوف نصية متراصة، مع صورة صغيرة تفتح في نافذة العرض. بدون بطاقات كبيرة.

```tsx
import React, { useMemo } from 'react';
import { Language, LightboxImageItem } from '../types';
import { HISTORY, SECTIONS } from '../data/content';
import { imgAlt, imgSrc } from '../data/images';
import { useLightbox } from '../context/LightboxContext';
import { Picture } from './Picture';
import { Reveal } from './Motion';
import { SectionHead } from './SectionHead';

interface TimelineSectionProps {
  lang: Language;
}

export const TimelineSection: React.FC<TimelineSectionProps> = React.memo(({ lang }) => {
  const isAr = lang === 'ar';
  const { openLightbox } = useLightbox();

  const slides: LightboxImageItem[] = useMemo(
    () =>
      HISTORY.map((e) => ({
        src: imgSrc(e.image),
        title: e.title[lang],
        caption: e.date[lang],
      })),
    [lang]
  );

  return (
    <section
      id="history"
      aria-labelledby="history-heading"
      className="bg-ivory px-5 py-16 sm:px-8 sm:py-24"
    >
      <div className="mx-auto max-w-6xl">
        <SectionHead
          id="history-heading"
          no="01"
          title={SECTIONS.history.title[lang]}
          lead={SECTIONS.history.lead[lang]}
          isAr={isAr}
        />

        <ol className="border-t border-line">
          {HISTORY.map((e, i) => (
            <li key={e.id}>
              <Reveal className="grid items-start gap-5 border-b border-line py-8 md:grid-cols-[11rem_1fr_9rem] md:gap-10 md:py-10">
                <div>
                  <span className="eyebrow">{String(i + 1).padStart(2, '0')}</span>
                  <p className="mt-1 text-sm text-ink-soft">{e.date[lang]}</p>
                </div>

                <div>
                  <h3 className="row-title">{e.title[lang]}</h3>
                  <p className="lead mt-3 max-w-2xl">{e.text[lang]}</p>
                </div>

                <button
                  type="button"
                  onClick={() => openLightbox(slides, i)}
                  aria-label={isAr ? `تكبير الصورة: ${e.title.ar}` : `Enlarge image: ${e.title.en}`}
                  className="group block aspect-[4/3] w-40 cursor-zoom-in overflow-hidden bg-[#efece6] md:w-full"
                >
                  <Picture
                    src={imgSrc(e.image)}
                    alt={imgAlt(e.image, lang)}
                    lang={lang}
                    className="h-full w-full transition-transform duration-700 group-hover:scale-[1.04]"
                  />
                </button>
              </Reveal>
            </li>
          ))}
        </ol>
      </div>
    </section>
  );
});
```

## الخطوة 5: `src/components/ArchitectureSection.tsx` (استبدال كامل)

ثلاثة صفوف نصية كبيرة. الأول مفتوح افتراضياً، والباقي يُفتح بالسهم.

```tsx
import React, { useMemo, useState } from 'react';
import { Language, LightboxImageItem } from '../types';
import { ARCHITECTURE, SECTIONS } from '../data/content';
import { imgAlt, imgSrc } from '../data/images';
import { useLightbox } from '../context/LightboxContext';
import { AccordionRow } from './AccordionRow';
import { Picture } from './Picture';
import { SectionHead } from './SectionHead';

interface ArchitectureSectionProps {
  lang: Language;
}

export const ArchitectureSection: React.FC<ArchitectureSectionProps> = React.memo(({ lang }) => {
  const isAr = lang === 'ar';
  const { openLightbox } = useLightbox();
  const [openId, setOpenId] = useState<string | null>(ARCHITECTURE[0].id);

  const slides: LightboxImageItem[] = useMemo(
    () =>
      ARCHITECTURE.map((a) => ({
        src: imgSrc(a.image),
        title: a.title[lang],
        caption: a.summary[lang],
      })),
    [lang]
  );

  return (
    <section
      id="architecture"
      aria-labelledby="architecture-heading"
      className="bg-white px-5 py-16 sm:px-8 sm:py-24"
    >
      <div className="mx-auto max-w-6xl">
        <SectionHead
          id="architecture-heading"
          no="02"
          title={SECTIONS.architecture.title[lang]}
          lead={SECTIONS.architecture.lead[lang]}
          isAr={isAr}
        />

        <div className="border-t border-line">
          {ARCHITECTURE.map((a, i) => (
            <AccordionRow
              key={a.id}
              id={`arch-${a.id}`}
              title={a.title[lang]}
              summary={a.summary[lang]}
              open={openId === a.id}
              onToggle={() => setOpenId(openId === a.id ? null : a.id)}
            >
              <div className="grid items-center gap-8 md:grid-cols-2 md:gap-12">
                <button
                  type="button"
                  onClick={() => openLightbox(slides, i)}
                  aria-label={isAr ? 'تكبير الصورة' : 'Enlarge image'}
                  className="block aspect-[4/3] w-full cursor-zoom-in overflow-hidden bg-[#efece6]"
                >
                  <Picture
                    src={imgSrc(a.image)}
                    alt={imgAlt(a.image, lang)}
                    lang={lang}
                    className="h-full w-full"
                  />
                </button>
                <p className="lead">{a.body[lang]}</p>
              </div>
            </AccordionRow>
          ))}
        </div>
      </div>
    </section>
  );
});
```
