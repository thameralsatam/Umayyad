## الخطوة 1: `src/components/PrayerWidget.tsx` (استبدال كامل)

```tsx
import React, { useEffect, useMemo, useState } from 'react';
import { Language } from '../types';
import { calculateDamascusPrayerTimes, formatTimeWithPeriod } from '../utils/prayerTimes';

interface PrayerWidgetProps {
  lang: Language;
}

export const PrayerWidget: React.FC<PrayerWidgetProps> = React.memo(({ lang }) => {
  const [stamp, setStamp] = useState<number>(() => Math.floor(Date.now() / 60000));

  useEffect(() => {
    const timer = setInterval(() => setStamp(Math.floor(Date.now() / 60000)), 15000);
    return () => clearInterval(timer);
  }, []);

  const data = useMemo(() => calculateDamascusPrayerTimes(new Date(), lang), [stamp, lang]);
  const isAr = lang === 'ar';

  const items = [
    { id: 'fajr', name: isAr ? 'الفجر' : 'Fajr', time: data.fajr },
    { id: 'sunrise', name: isAr ? 'الشروق' : 'Sunrise', time: data.sunrise },
    { id: 'dhuhr', name: isAr ? 'الظهر' : 'Dhuhr', time: data.dhuhr },
    { id: 'asr', name: isAr ? 'العصر' : 'Asr', time: data.asr },
    { id: 'maghrib', name: isAr ? 'المغرب' : 'Maghrib', time: data.maghrib },
    { id: 'isha', name: isAr ? 'العشاء' : 'Isha', time: data.isha },
  ];

  return (
    <div id="prayer-times-widget" className="card-soft p-4 sm:p-6">
      <div className="flex flex-wrap items-center justify-between gap-2 px-2 pb-4">
        <h3 className="text-base font-medium text-ink">
          {isAr ? 'مواقيت الصلاة في دمشق' : 'Prayer times in Damascus'}
        </h3>
        <span className="rounded-full bg-sand px-3 py-1 text-sm text-ink-soft">{data.hijriDate}</span>
      </div>

      <dl className="grid grid-cols-3 gap-2 md:grid-cols-6">
        {items.map((p) => {
          const next = data.nextPrayer?.id === p.id;
          return (
            <div
              key={p.id}
              className={`rounded-2xl px-1 py-4 text-center transition-colors ${
                next ? 'bg-ivory ring-1 ring-gold' : 'bg-transparent'
              }`}
            >
              <dt className={`text-sm ${next ? 'font-medium text-gold-deep' : 'text-ink-soft'}`}>{p.name}</dt>
              <dd className="mt-1 text-xl font-light tabular-nums text-ink sm:text-2xl">
                {formatTimeWithPeriod(p.time, lang)}
              </dd>
            </div>
          );
        })}
      </dl>
    </div>
  );
});
```

## الخطوة 6: `src/components/PhotoGallerySection.tsx` (استبدال كامل)

شريط أفقي بأحجام متنوعة وحواف دائرية. كل صورة تنفتح بقناع، وتتنفس (تكبير وتصغير خفيف) مع التمرير.

```tsx
import React, { useMemo, useRef } from 'react';
import { motion, useReducedMotion, useScroll, useTransform } from 'motion/react';
import { ChevronLeft, ChevronRight } from 'lucide-react';
import { Language, LightboxImageItem } from '../types';
import { SECTIONS } from '../data/content';
import { GALLERY_KEYS, IMAGES, ImageKey, imgSrc } from '../data/images';
import { useLightbox } from '../context/LightboxContext';
import { Picture } from './Picture';
import { SectionHead } from './SectionHead';

interface PhotoGallerySectionProps {
  lang: Language;
}

const SIZES = [
  'h-72 w-52 sm:h-96 sm:w-64',
  'h-48 w-72 sm:h-64 sm:w-96',
  'h-60 w-60 sm:h-72 sm:w-72',
  'h-64 w-52 sm:h-80 sm:w-60',
];

interface ItemProps {
  imageKey: ImageKey;
  index: number;
  lang: Language;
  onOpen: (i: number) => void;
  scroller: React.RefObject<HTMLDivElement | null>;
}

const GalleryItem: React.FC<ItemProps> = ({ imageKey, index, lang, onOpen, scroller }) => {
  const ref = useRef<HTMLElement>(null);
  const reduce = useReducedMotion();
  const { scrollYProgress } = useScroll({ target: ref, offset: ['start end', 'end start'] });
  const scale = useTransform(scrollYProgress, [0, 0.5, 1], [0.95, 1, 0.98]);

  return (
    <motion.figure ref={ref} style={reduce ? undefined : { scale }} className="shrink-0 snap-start">
      <button
        type="button"
        onClick={() => onOpen(index)}
        aria-label={IMAGES[imageKey][lang]}
        className={`group block cursor-zoom-in overflow-hidden rounded-[2rem] bg-[#efece6] shadow-[0_18px_40px_-24px_rgba(0,38,35,0.4)] ${SIZES[index % SIZES.length]}`}
      >
        <motion.div
          className="h-full w-full"
          initial={reduce ? false : { clipPath: 'inset(0 0 100% 0)' }}
          whileInView={{ clipPath: 'inset(0 0 0% 0)' }}
          viewport={{ root: scroller, once: true, amount: 0.25 }}
          transition={{ duration: 0.9, delay: (index % 3) * 0.08, ease: [0.22, 1, 0.36, 1] }}
        >
          <Picture
            src={imgSrc(imageKey)}
            alt={IMAGES[imageKey][lang]}
            lang={lang}
            className="h-full w-full transition-transform duration-700 group-hover:scale-[1.05]"
          />
        </motion.div>
      </button>
      <figcaption className="mt-3 px-2 text-sm text-ink-soft">{IMAGES[imageKey][lang]}</figcaption>
    </motion.figure>
  );
};

export const PhotoGallerySection: React.FC<PhotoGallerySectionProps> = React.memo(({ lang }) => {
  const isAr = lang === 'ar';
  const { openLightbox } = useLightbox();
  const scroller = useRef<HTMLDivElement>(null);

  const slides: LightboxImageItem[] = useMemo(
    () => GALLERY_KEYS.map((key) => ({ src: imgSrc(key), title: IMAGES[key][lang] })),
    [lang]
  );

  const scrollByStep = (next: boolean) => {
    const el = scroller.current;
    if (!el) return;
    const amount = el.clientWidth * 0.8;
    const toRight = isAr ? !next : next;
    el.scrollBy({ left: toRight ? amount : -amount, behavior: 'smooth' });
  };

  return (
    <section id="gallery" aria-labelledby="gallery-heading" className="overflow-hidden bg-white py-16 sm:py-24">
      <div className="mx-auto max-w-6xl px-5 sm:px-8">
        <SectionHead
          id="gallery-heading"
          no="04"
          title={SECTIONS.gallery.title[lang]}
          lead={SECTIONS.gallery.lead[lang]}
          isAr={isAr}
        />
      </div>

      <div
        ref={scroller}
        tabIndex={0}
        aria-label={isAr ? 'صور المعرض، مرّر أفقياً' : 'Gallery photos, scroll horizontally'}
        className="flex snap-x snap-mandatory items-end gap-4 overflow-x-auto px-5 pb-6 pt-4 sm:gap-6 sm:px-8"
      >
        {GALLERY_KEYS.map((key, i) => (
          <GalleryItem
            key={key}
            imageKey={key}
            index={i}
            lang={lang}
            scroller={scroller}
            onOpen={(n) => openLightbox(slides, n)}
          />
        ))}
      </div>

      <div className="mx-auto mt-4 flex max-w-6xl gap-3 px-5 sm:px-8">
        <button
          type="button"
          onClick={() => scrollByStep(false)}
          aria-label={isAr ? 'السابق' : 'Previous'}
          className="grid h-12 w-12 cursor-pointer place-items-center rounded-full border border-line bg-white text-ink transition-colors hover:border-gold hover:bg-ivory"
        >
          {isAr ? (
            <ChevronRight className="h-5 w-5" strokeWidth={1.5} />
          ) : (
            <ChevronLeft className="h-5 w-5" strokeWidth={1.5} />
          )}
        </button>
        <button
          type="button"
          onClick={() => scrollByStep(true)}
          aria-label={isAr ? 'التالي' : 'Next'}
          className="grid h-12 w-12 cursor-pointer place-items-center rounded-full border border-line bg-white text-ink transition-colors hover:border-gold hover:bg-ivory"
        >
          {isAr ? (
            <ChevronLeft className="h-5 w-5" strokeWidth={1.5} />
          ) : (
            <ChevronRight className="h-5 w-5" strokeWidth={1.5} />
          )}
        </button>
      </div>
    </section>
  );
});

import { CONDUCT, GATES, HOURS, SECTIONS, VISIT_LABELS } from '../data/content';
import { AccordionRow } from './AccordionRow';
import { Reveal } from './Motion';
import { SectionHead } from './SectionHead';

interface VisitSectionProps {
  lang: Language;
}

export const VisitSection: React.FC<VisitSectionProps> = React.memo(({ lang }) => {
  const isAr = lang === 'ar';
  const [openId, setOpenId] = useState<string | null>(null);

  return (
    <section id="visit" aria-labelledby="visit-heading" className="bg-ivory px-5 py-16 sm:px-8 sm:py-24">
      <div className="mx-auto max-w-6xl">
        <SectionHead
          id="visit-heading"
          no="05"
          title={SECTIONS.visit.title[lang]}
          lead={SECTIONS.visit.lead[lang]}
          isAr={isAr}
        />

        <h3 className="sr-only">{VISIT_LABELS.gates[lang]}</h3>
        <div className="space-y-4">
          {GATES.map((g, i) => (
            <Reveal key={g.id} x={(i % 2 === 0 ? -1 : 1) * 48 * (isAr ? -1 : 1)}>
              <AccordionRow
                id={`gate-${g.id}`}
                eyebrow={g.direction[lang]}
                title={g.name[lang]}
                summary={g.summary[lang]}
                open={openId === g.id}
                onToggle={() => setOpenId(openId === g.id ? null : g.id)}
              >
                <p className="lead max-w-2xl">{g.more[lang]}</p>
              </AccordionRow>
            </Reveal>
          ))}
        </div>

        <div className="mt-16 sm:mt-20">
          <Reveal>
            <h3 className="row-title mb-8">{VISIT_LABELS.conduct[lang]}</h3>
          </Reveal>
          <ul className="grid gap-4 sm:grid-cols-2 sm:gap-6 lg:grid-cols-4">
            {CONDUCT.map((c, i) => (
              <li key={c.id}>
                <Reveal delay={i * 0.08} className="h-full">
                  <div className="card-soft h-full p-6 sm:p-7">
                    <h4 className="text-lg font-medium text-ink">{c.title[lang]}</h4>
                    <p className="mt-2 text-base font-light leading-8 text-ink-soft">{c.text[lang]}</p>
                  </div>
                </Reveal>
              </li>
            ))}
          </ul>
        </div>

        <Reveal className="mt-10 sm:mt-14">
          <div className="card-soft flex flex-col gap-4 p-6 sm:flex-row sm:items-center sm:gap-8 sm:p-8">
            <span className="pill w-fit border-gold/50 bg-sand text-gold-deep">{VISIT_LABELS.hours[lang]}</span>
            <p className="lead">{HOURS[lang]}</p>
          </div>
        </Reveal>
      </div>
    </section>
  );
});
```

## الخطوة 8: الفحص النهائي

1. شغّل `npm run lint` وأصلح الأخطاء بأقل تعديل (استيراد ناقص أو خطأ إملائي). لا تغيّر التصميم ولا النصوص.
2. شغّل `npm run build`. يجب أن ينجح.
3. ابحث في `src/` عن أي استيراد من ملفات محذوفة (`SafeImage`، `MosqueProgrammaticArtwork`، `SectionTransitionWrapper`، `AnimatedNumber`، `FormattedText`، `ParallaxStatsSection`) وعن `MOSQUE_IMAGES` أو `MOSAIC_ARTWORKS` أو `TIMELINE_EVENTS` أو `HERO_CONTENT` أو `PHOTO_GALLERY_ITEMS` أو `ETIQUETTE_GUIDELINES`. إن وجدت شيئاً أخبر المستخدم ولا تحذفه بنفسك.
4. أجب بـ "نعم" أو "لا" فقط بعد التحقق الفعلي:
   - لا يوجد وضع داكن ولا أي `dark:` في الكود
   - العدّاد موجود في الرئيسية فقط، ولا توجد أرقام أخرى متحركة
   - الحركة الموازية موجودة في موضعين فقط: صورة الرئيسية وصورة الفسيفساء
   - خلفية التذييل وحده خضراء داكنة
   - لا توجد أي أرقام ارتفاع أو طول أو مساحة في المحتوى
   - الأزرار والتبويبات والوسوم كلها كبسولات، والبطاقات بحواف دائرية
   - التبديل بين العربية والإنجليزية يقلب اتجاه حركات الانزلاق الجانبية
   - نافذة عرض الصور تفتح من كل الأقسام
5. اكتب للمستخدم تقريراً من خمسة أسطر كحد أقصى.
