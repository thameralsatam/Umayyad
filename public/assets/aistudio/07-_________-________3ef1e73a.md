4. في النهاية فقط: `npm run lint` ثم `npm run build`، وأصلح الأخطاء بأقل تعديل دون تغيير التصميم.
5. اكتب تقريراً من خمسة أسطر كحد أقصى.

```tsx

import { ChevronLeft, ChevronRight } from 'lucide-react';
import { Language, LightboxImageItem } from '../types';
import { SECTIONS } from '../data/content';
import { GALLERY_KEYS, IMAGES, ImageKey, imgSrc } from '../data/images';
import { useLightbox } from '../context/LightboxContext';
import { Picture } from './Picture';
import { SectionHead } from './SectionHead';

interface PhotoGallerySectionProps {
  lang: Language;
}

interface ItemProps {
  imageKey: ImageKey;
  index: number;
  lang: Language;
  onOpen: (i: number) => void;
  scroller: React.RefObject<HTMLDivElement | null>;
}

const GalleryItem: React.FC<ItemProps> = ({ imageKey, index, lang, onOpen, scroller }) => {
  const ref = useRef<HTMLElement>(null);
  const reduce = useReducedMotion();
  const { scrollYProgress } = useScroll({ target: ref, offset: ['start end', 'end start'] });
  const scale = useTransform(scrollYProgress, [0, 0.5, 1], [0.95, 1, 0.98]);

  return (
    <motion.figure ref={ref} style={reduce ? undefined : { scale }} className="w-60 shrink-0 snap-start sm:w-72">
      <button
        type="button"
        onClick={() => onOpen(index)}
        aria-label={IMAGES[imageKey][lang]}

          className="h-full w-full"
          initial={reduce ? false : { clipPath: 'inset(0 0 100% 0)' }}
          whileInView={{ clipPath: 'inset(0 0 0% 0)' }}
          viewport={{ root: scroller, once: true, amount: 0.25 }}
          transition={{ duration: 0.9, delay: (index % 3) * 0.08, ease: [0.22, 1, 0.36, 1] }}
        >
          <Picture
            src={imgSrc(imageKey)}
            alt={IMAGES[imageKey][lang]}
            lang={lang}
            className="h-full w-full transition-transform duration-700 group-hover:scale-[1.05]"
          />
        </motion.div>
      </button>
      <figcaption className="mt-3 px-2 text-sm text-ink-soft">{IMAGES[imageKey][lang]}</figcaption>
    </motion.figure>
  );

          lead={SECTIONS.gallery.lead[lang]}
```

## الخطوة 8: `src/components/UniversalLightbox.tsx` (استبدال كامل)

نافذة عرض عصرية: خلفية ضبابية، أزرار دائرية زجاجية، صورة بحواف دائرية، نقاط تنقل، وبطاقة وصف.

```tsx
import React, { useCallback, useEffect, useRef, useState } from 'react';
import { createPortal } from 'react-dom';
import { motion } from 'motion/react';
import { ChevronLeft, ChevronRight, X } from 'lucide-react';
import { Language, LightboxImageItem } from '../types';

interface UniversalLightboxProps {
  images: LightboxImageItem[];
  currentIndex: number;
  onClose: () => void;
  onNavigate: (index: number) => void;
  lang: Language;
}

const glass =
  'grid cursor-pointer place-items-center rounded-full border border-white/20 bg-white/10 text-white backdrop-blur-md transition-colors hover:bg-white/20';

export const UniversalLightbox: React.FC<UniversalLightboxProps> = ({
  images,
  currentIndex,
  onClose,
  onNavigate,
  lang,
}) => {
  const isAr = lang === 'ar';
  const count = images.length;
  const item = images[currentIndex] || images[0];

  const [zoomed, setZoomed] = useState(false);
  const [failed, setFailed] = useState(false);
  const touchX = useRef<number | null>(null);

  const go = useCallback(
    (step: number) => {
      if (count <= 1) return;
      onNavigate((currentIndex + step + count) % count);
    },
    [count, currentIndex, onNavigate]
  );

  useEffect(() => {
    setZoomed(false);
    setFailed(false);
  }, [currentIndex]);

  useEffect(() => {
    const previous = document.body.style.overflow;
    document.body.style.overflow = 'hidden';
    return () => {
      document.body.style.overflow = previous;
    };
  }, []);

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onClose();
      else if (e.key === 'ArrowRight') go(isAr ? -1 : 1);
      else if (e.key === 'ArrowLeft') go(isAr ? 1 : -1);
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [go, isAr, onClose]);

  if (!item) return null;

  const onTouchStart = (e: React.TouchEvent) => {
    touchX.current = e.touches[0].clientX;
  };

  const onTouchEnd = (e: React.TouchEvent) => {
    if (touchX.current === null || zoomed) return;
    const delta = e.changedTouches[0].clientX - touchX.current;
    touchX.current = null;
    if (Math.abs(delta) < 50) return;
    const dir = delta < 0 ? 1 : -1;
    go(isAr ? -dir : dir);
  };

  const pad = (n: number) => String(n).padStart(2, '0');

  return createPortal(
    <motion.div
      role="dialog"
      aria-modal="true"
      aria-label={item.title}
      initial={{ opacity: 0 }}
      animate={{ opacity: 1 }}
      transition={{ duration: 0.3 }}
      className="fixed inset-0 z-[9999] flex flex-col bg-[#001a18]/90 text-white backdrop-blur-xl"
      onClick={onClose}
    >
      <div className="flex items-center justify-between px-4 py-4 sm:px-8" onClick={(e) => e.stopPropagation()}>
        <span className="rounded-full border border-white/20 bg-white/10 px-4 py-1.5 text-sm tabular-nums backdrop-blur-md">
          {pad(currentIndex + 1)} / {pad(count)}
        </span>
        <button
          type="button"
          onClick={onClose}
          aria-label={isAr ? 'إغلاق' : 'Close'}
          className={`h-11 w-11 ${glass}`}
        >
          <X className="h-5 w-5" strokeWidth={1.5} />
        </button>
      </div>

      <div className="relative min-h-0 flex-1" onClick={(e) => e.stopPropagation()}>
        <div
          className={`absolute inset-0 flex px-4 sm:px-24 ${zoomed ? 'overflow-auto' : 'overflow-hidden'}`}
          onTouchStart={onTouchStart}
          onTouchEnd={onTouchEnd}
          onClick={(e) => {
            if (e.target === e.currentTarget) onClose();
          }}
        >
          {failed ? (
            <p className="m-auto text-white/70">{isAr ? 'الصورة غير متوفرة' : 'Image unavailable'}</p>
          ) : (
            <motion.img
              key={item.src}
              initial={{ opacity: 0, scale: 0.97 }}
              animate={{ opacity: 1, scale: 1 }}
              transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}
              src={item.src}
              alt={item.title}
              draggable={false}
              onError={() => setFailed(true)}
              onClick={() => setZoomed((z) => !z)}
              className={`m-auto select-none rounded-[1.75rem] shadow-[0_30px_80px_-20px_rgba(0,0,0,0.6)] ${
                zoomed
                  ? 'w-[220%] max-w-none cursor-zoom-out'
                  : 'max-h-full max-w-full cursor-zoom-in object-contain'
              }`}
            />
          )}
        </div>

        {count > 1 && (
          <>
            <button
              type="button"
              onClick={() => go(isAr ? 1 : -1)}
              aria-label={isAr ? 'التالي' : 'Previous'}
              className={`absolute left-3 top-1/2 h-12 w-12 -translate-y-1/2 sm:left-6 ${glass}`}
            >
              <ChevronLeft className="h-5 w-5" strokeWidth={1.5} />
            </button>
            <button
              type="button"
              onClick={() => go(isAr ? -1 : 1)}
              aria-label={isAr ? 'السابق' : 'Next'}
              className={`absolute right-3 top-1/2 h-12 w-12 -translate-y-1/2 sm:right-6 ${glass}`}
            >
              <ChevronRight className="h-5 w-5" strokeWidth={1.5} />
            </button>
          </>
        )}
      </div>

      <div className="flex flex-col items-center gap-4 px-4 pb-6 pt-4 sm:px-8" onClick={(e) => e.stopPropagation()}>
        <div className="max-w-2xl rounded-3xl border border-white/15 bg-white/10 px-6 py-4 text-center backdrop-blur-md">
          <h3 className="text-lg font-light sm:text-xl">{item.title}</h3>
          {item.caption && <p className="mt-1 text-sm font-light text-white/70">{item.caption}</p>}
        </div>

        {count > 1 && count <= 12 && (
          <div className="flex items-center gap-2" aria-hidden="true">
            {images.map((_, i) => (
              <button
                key={i}
                type="button"
                tabIndex={-1}
                onClick={() => onNavigate(i)}
                className={`h-2 cursor-pointer rounded-full transition-all duration-300 ${
                  i === currentIndex ? 'w-7 bg-[#b9a779]' : 'w-2 bg-white/35'
                }`}
              />
            ))}
          </div>
        )}
      </div>
    </motion.div>,
    document.body
  );
};
```

الأزرار الجانبية موضوعة بحسب الشاشة (يسار ويمين) وليس بحسب اللغة، فلا تغيّر `aria-label` فيهما.

## الخطوة 9: الفحص النهائي

1. `npm run lint` ثم `npm run build`. أصلح الأخطاء بأقل تعديل دون تغيير التصميم.
2. أجب بـ "نعم" أو "لا" فقط بعد التحقق الفعلي:
   - العنوان في وسط الشاشة الأولى وبوزن متوسط
   - لا يظهر في الشاشة الأولى أي شيء غير العنوان والوصف وزر التمرير
   - كل صور المعرض بنفس الأبعاد
   - الصورة الرئيسية تفتح في نافذة العرض عند الضغط
   - الحقب الأربع لها نفس نوع الظهور، وكذلك المداخل الأربعة
   - نافذة العرض بأزرار دائرية وخلفية ضبابية
   - لا يوجد `dark:` في أي ملف
3. تقرير من خمسة أسطر كحد أقصى.
